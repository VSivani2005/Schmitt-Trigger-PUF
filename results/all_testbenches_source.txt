`timescale 1ns / 1ps

module multi_gpio_measurement_tb;

    reg clk = 0;
    reg [9:0] gpio_inputs = 10'b0;

    reg [3:0] channel_select = 0;
    reg req_meas = 0;

    wire selected_pin;
    wire feedback_out;

    wire [23:0] measured_period;
    wire ack_meas;
    wire busy;

    reg [239:0] period_values = 240'd0;
    wire [44:0] response;

    integer i;
    integer j;
    integer errors;

    // 10 MHz simulation clock
    always #50 clk = ~clk;

    // Ten GPIO signals with different simulated timings
   always #525  gpio_inputs[0] = ~gpio_inputs[0];
   always #725  gpio_inputs[1] = ~gpio_inputs[1];
   always #625  gpio_inputs[2] = ~gpio_inputs[2];
   always #1125 gpio_inputs[3] = ~gpio_inputs[3];
   always #1025 gpio_inputs[4] = ~gpio_inputs[4];
   always #1525 gpio_inputs[5] = ~gpio_inputs[5];
   always #1325 gpio_inputs[6] = ~gpio_inputs[6];
   always #2125 gpio_inputs[7] = ~gpio_inputs[7];
   always #1925 gpio_inputs[8] = ~gpio_inputs[8];
   always #2325 gpio_inputs[9] = ~gpio_inputs[9];

    // GPIO selection
    schmitt_mux mux_inst (
        .gpio_inputs(gpio_inputs),
        .channel_select(channel_select),
        .selected_pin(selected_pin)
    );

    // Period measurement
    period_measurement measurement_inst (
        .clk_100mhz(clk),
        .req_meas(req_meas),
        .pin_selected(selected_pin),
        .pin_out(feedback_out),
        .data_out(measured_period),
        .ack_meas(ack_meas),
        .busy(busy)
    );

    // Pairwise response generation
    pairwise_response response_inst (
        .period_values(period_values),
        .response(response)
    );

    // Measure one selected GPIO
    task measure_channel;
        input integer idx;

        begin
            @(negedge clk);

            channel_select = idx;

            // Allow the selected signal to settle
            repeat (2) @(negedge clk);

            req_meas = 1'b1;

            // Wait for the measurement to complete
            wait (ack_meas == 1'b1);

            // Store the measured period
            period_values[idx*24 +: 24] = measured_period;

            @(negedge clk);
            req_meas = 1'b0;

            // Allow the measurement module to return to IDLE
            repeat (2) @(negedge clk);
        end
    endtask

    initial begin

        $display("------------------------------------------");
        $display("MULTI-GPIO PUF MEASUREMENT TEST");
        $display("------------------------------------------");

        // Measure all ten GPIO inputs
        for (i = 0; i < 10; i = i + 1) begin

            measure_channel(i);

            $display(
                "GPIO %0d measured period = %0d cycles",
                i, period_values[i*24 +: 24]
            );

        end

        #1;

        $display("\n------------------------------------------");
        $display("MEASURED PERIOD SUMMARY");
        $display("------------------------------------------");

        errors = 0;

        // Check for zero or duplicate measurements
        for (i = 0; i < 10; i = i + 1) begin

            if (period_values[i*24 +: 24] == 0) begin
                $display("ERROR: GPIO %0d has zero measurement", i);
                errors = errors + 1;
            end

            for (j = i + 1; j < 10; j = j + 1) begin

                if (period_values[i*24 +: 24] ==
                    period_values[j*24 +: 24]) begin

                    $display(
                        "WARNING: GPIO %0d and GPIO %0d have equal periods",
                        i, j
                    );

                    errors = errors + 1;
                end

            end

        end

        $display("\nGenerated 45-bit response:");
        $display("%045b", response);

        $display("\n------------------------------------------");
        $display("FINAL VERIFICATION");
        $display("------------------------------------------");

        if (errors == 0) begin
            $display("PASS: All ten GPIO periods are non-zero and distinct");
            $display("PASS: Pairwise response generated");
        end
        else begin
            $display("FAIL: Measurement errors = %0d", errors);
        end

        $display("------------------------------------------");

        $finish;

    end

    // Simulation watchdog
    initial begin
        #200000;
        $display("ERROR: Simulation timeout");
        $finish;
    end

endmodule

module pairwise_puf_tb;

    reg [239:0] period_values;
    wire [44:0] response;

    integer i;
    integer j;
    integer k;
    integer ones_count;
    integer zeros_count;
    integer errors;

    reg expected_bit;

    pairwise_response uut (
        .period_values(period_values),
        .response(response)
    );

    initial begin

        $display("------------------------------------------");
        $display("10-GPIO PAIRWISE PUF RESPONSE TEST");
        $display("------------------------------------------");

        // Simulated period values in nanoseconds
        period_values[0*24 +: 24] = 24'd100;
        period_values[1*24 +: 24] = 24'd140;
        period_values[2*24 +: 24] = 24'd120;
        period_values[3*24 +: 24] = 24'd180;
        period_values[4*24 +: 24] = 24'd160;
        period_values[5*24 +: 24] = 24'd220;
        period_values[6*24 +: 24] = 24'd200;
        period_values[7*24 +: 24] = 24'd260;
        period_values[8*24 +: 24] = 24'd240;
        period_values[9*24 +: 24] = 24'd280;

        #10;

        k = 0;
        ones_count = 0;
        zeros_count = 0;
        errors = 0;

        $display("\nGPIO period measurements:");

        for (i = 0; i < 10; i = i + 1) begin
            $display("GPIO %0d = %0d ns",
                     i, period_values[i*24 +: 24]);
        end

        $display("\nPairwise comparisons:");

        for (i = 0; i < 10; i = i + 1) begin

            for (j = i + 1; j < 10; j = j + 1) begin

                if (period_values[i*24 +: 24] >
                    period_values[j*24 +: 24])
                    expected_bit = 1'b1;
                else
                    expected_bit = 1'b0;

                $display("GPIO %0d vs GPIO %0d : Response = %b",
                         i, j, response[k]);

                if (response[k] !== expected_bit) begin
                    $display("ERROR: Pair %0d,%0d", i, j);
                    errors = errors + 1;
                end

                if (response[k] == 1'b1)
                    ones_count = ones_count + 1;
                else
                    zeros_count = zeros_count + 1;

                k = k + 1;

            end

        end

        $display("\n------------------------------------------");
        $display("PAIRWISE RESPONSE SUMMARY");
        $display("------------------------------------------");

        $display("Total comparisons = %0d", k);
        $display("Response 1 count = %0d", ones_count);
        $display("Response 0 count = %0d", zeros_count);
        $display("Generated response = %045b", response);

        if (k == 45 && errors == 0)
            $display("PASS: All 45 pairwise comparisons verified");
        else
            $display("FAIL: Pairwise comparison errors = %0d",
                     errors);

        $display("------------------------------------------");

        $finish;

    end

endmodule

========== period_measurement_tb.v ==========
`timescale 1ns / 1ps

module period_measurement_tb;

    reg clk_100mhz;
    reg rst;
    reg req_meas;
    reg pin_selected;
    reg completed;

    wire pin_out;
    wire [23:0] data_out;
    wire ack_meas;
    wire busy;

    period_measurement uut (
        .clk_100mhz(clk_100mhz),
        .rst(rst),
        .req_meas(req_meas),
        .pin_selected(pin_selected),
        .pin_out(pin_out),
        .data_out(data_out),
        .ack_meas(ack_meas),
        .busy(busy)
    );

    always #5 clk_100mhz = ~clk_100mhz;

    // Generate a 1 us oscillation period
    initial begin
        #100;
        forever begin
            #500 pin_selected = 1'b1;
            #500 pin_selected = 1'b0;
        end
    end

    // Independent timeout
    initial begin
        #10000;
        if (!completed) begin
            $display("FAIL: Measurement timeout");
            $finish;
        end
    end

    initial begin
        clk_100mhz = 0;
        rst = 1;
        req_meas = 0;
        pin_selected = 0;
        completed = 0;

        $display("----------------------------------");
        $display("GPIO SCHMITT PUF - PERIOD TEST");
        $display("----------------------------------");

        #50;
        rst = 0;

        #150;
        req_meas = 1;
        $display("Measurement request started");

        wait (ack_meas == 1'b1);
        #1;

        completed = 1;

        $display("Measured clock cycles = %d", data_out);
        $display("Measurement ACK received");

        @(negedge clk_100mhz);
        req_meas = 0;

        #100;

        if (data_out == 24'd100)
            $display("PASS: Expected period measurement obtained");
        else
            $display("FAIL: Expected 100, received %d", data_out);

        $display("----------------------------------");
        $finish;
    end

endmodule

module puf_reliability_tb;

    reg clk = 0;
    reg [9:0] gpio_inputs = 10'b0;

    reg [3:0] channel_select = 0;
    reg req_meas = 0;

    wire selected_pin;
    wire feedback_out;

    wire [23:0] measured_period;
    wire ack_meas;
    wire busy;

    reg [239:0] period_values = 240'd0;
    wire [44:0] response;

    reg [44:0] reference_response;

    integer round_idx;
    integer gpio_idx;
    integer bit_idx;
    integer diff_bits;
    integer stability_x100;
    integer sum_stability_x100;
    integer average_stability_x100;

    // 10 MHz clock
    always #50 clk = ~clk;

    // Simulated GPIO signals
    always #525  gpio_inputs[0] = ~gpio_inputs[0];
    always #725  gpio_inputs[1] = ~gpio_inputs[1];
    always #625  gpio_inputs[2] = ~gpio_inputs[2];
    always #1125 gpio_inputs[3] = ~gpio_inputs[3];
    always #1025 gpio_inputs[4] = ~gpio_inputs[4];
    always #1525 gpio_inputs[5] = ~gpio_inputs[5];
    always #1325 gpio_inputs[6] = ~gpio_inputs[6];
    always #2125 gpio_inputs[7] = ~gpio_inputs[7];
    always #1925 gpio_inputs[8] = ~gpio_inputs[8];
    always #2325 gpio_inputs[9] = ~gpio_inputs[9];

    // GPIO multiplexer
    schmitt_mux mux_inst (
        .gpio_inputs(gpio_inputs),
        .channel_select(channel_select),
        .selected_pin(selected_pin)
    );

    // Period measurement
    period_measurement measurement_inst (
        .clk_100mhz(clk),
        .req_meas(req_meas),
        .pin_selected(selected_pin),
        .pin_out(feedback_out),
        .data_out(measured_period),
        .ack_meas(ack_meas),
        .busy(busy)
    );

    // Pairwise response generator
    pairwise_response response_inst (
        .period_values(period_values),
        .response(response)
    );

    // Measure one GPIO
    task measure_channel;
        input integer idx;

        begin
            @(negedge clk);
            channel_select = idx;

            repeat (2) @(negedge clk);

            req_meas = 1'b1;

            wait (ack_meas == 1'b1);

            period_values[idx*24 +: 24] = measured_period;

            @(negedge clk);
            req_meas = 1'b0;

            repeat (2) @(negedge clk);
        end
    endtask

    initial begin

        $display("==========================================");
        $display("PUF RESPONSE RELIABILITY TEST");
        $display("==========================================");

        sum_stability_x100 = 0;
        reference_response = 45'd0;

        // Perform 10 measurement rounds
        for (round_idx = 0; round_idx < 10;
             round_idx = round_idx + 1) begin

            period_values = 240'd0;

            $display("\nROUND %0d", round_idx + 1);

            // Measure all ten GPIOs
            for (gpio_idx = 0; gpio_idx < 10;
                 gpio_idx = gpio_idx + 1) begin

                measure_channel(gpio_idx);

                $display("GPIO %0d = %0d cycles",
                    gpio_idx,
                    period_values[gpio_idx*24 +: 24]);

            end

            // Allow combinational response to settle
            #1;

            $display("Response = %045b", response);

            if (round_idx == 0) begin

                reference_response = response;

                $display("Reference response recorded");

            end
            else begin

                diff_bits = 0;

                // Compare with first-round response
                for (bit_idx = 0; bit_idx < 45;
                     bit_idx = bit_idx + 1) begin

                    if (response[bit_idx] !==
                        reference_response[bit_idx])
                        diff_bits = diff_bits + 1;

                end

                stability_x100 =
                    ((45 - diff_bits) * 10000) / 45;

                sum_stability_x100 =
                    sum_stability_x100 + stability_x100;

                $display("Different bits = %0d", diff_bits);

                $display("Round stability = %0d.%02d%%",
                    stability_x100 / 100,
                    stability_x100 % 100);

            end

        end

        // Average over rounds 2 through 10
        average_stability_x100 =
            sum_stability_x100 / 9;

        $display("\n==========================================");
        $display("RELIABILITY TEST SUMMARY");
        $display("==========================================");

        $display("Measurement rounds = 10");
        $display("GPIO inputs = 10");
        $display("Response bits = 45");
        $display("Repeated comparisons = 9");

        $display("Average response stability = %0d.%02d%%",
            average_stability_x100 / 100,
            average_stability_x100 % 100);

        $display("==========================================");
        $display("PUF RELIABILITY SIMULATION COMPLETED");
        $display("==========================================");

        $finish;

    end

    // Watchdog
    initial begin
        #2000000;
        $display("ERROR: Simulation timeout");
        $finish;
    end

endmodule

module puf_top_tb;

    reg clk;
    reg rst;

    reg uart_rx_pin;
    wire uart_tx_pin;

    reg [9:0] gpio_inputs;

    wire [7:0] monitor_data;
    wire monitor_dv;

    integer rx_count;
    integer errors;

    reg [7:0] result_byte1;
    reg [7:0] result_byte2;
    reg [7:0] result_byte3;

    localparam integer BIT_TIME = 8680;

    // Device under test
    puf_top uut (
        .clk(clk),
        .rst(rst),
        .uart_rx_pin(uart_rx_pin),
        .uart_tx_pin(uart_tx_pin),
        .gpio_inputs(gpio_inputs)
    );

    // UART receiver used to monitor the DUT's serial output
    uart_rx monitor (
        .clk(clk),
        .rst(rst),
        .rx_serial(uart_tx_pin),
        .rx_data(monitor_data),
        .rx_dv(monitor_dv)
    );

    // 10 MHz clock
    always #5 clk = ~clk;

    // Simulated oscillating GPIO 7
    always #500 gpio_inputs[7] = ~gpio_inputs[7];

    // Send one byte to the FPGA using UART
    task send_uart_byte;
        input [7:0] data;
        integer b;
        begin
            uart_rx_pin = 1'b0;
            #(BIT_TIME);

            for (b = 0; b < 8; b = b + 1) begin
                uart_rx_pin = data[b];
                #(BIT_TIME);
            end

            uart_rx_pin = 1'b1;
            #(BIT_TIME);
        end
    endtask

    // Monitor bytes returned by the FPGA
    always @(posedge clk) begin
        if (!rst && monitor_dv) begin

            rx_count = rx_count + 1;

            case (rx_count)

                1: result_byte1 = monitor_data;
                2: result_byte2 = monitor_data;
                3: result_byte3 = monitor_data;

            endcase

            $display("FPGA returned byte %0d = 0x%02h",
                     rx_count, monitor_data);
        end
    end

    initial begin

        clk = 0;
        rst = 1;

        uart_rx_pin = 1'b1;
        gpio_inputs = 10'b0000000000;

        rx_count = 0;
        errors = 0;

        result_byte1 = 0;
        result_byte2 = 0;
        result_byte3 = 0;

        $display("----------------------------------");
        $display("COMPLETE SCHMITT PUF RTL TEST");
        $display("----------------------------------");

        // Reset the design
        repeat (10) @(negedge clk);
        rst = 0;

        // Send command byte
        $display("Sending command: 10");
        send_uart_byte(8'd10);

        // Send GPIO channel
        $display("Sending GPIO selection: 7");
        send_uart_byte(8'd7);

        // Wait for all three result bytes
        wait (rx_count == 3);

        #200;

        $display("----------------------------------");
        $display("RESULT VERIFICATION");
        $display("----------------------------------");

        $display("Received 24-bit result = 0x%02h%02h%02h",
                 result_byte1, result_byte2, result_byte3);

        if ({result_byte1, result_byte2, result_byte3} != 24'd0)
            $display("PASS: Non-zero measurement returned");
        else begin
            $display("FAIL: Measurement result is zero");
            errors = errors + 1;
        end

        if (uut.channel_select == 4'd7)
            $display("PASS: Correct GPIO channel selected");
        else begin
            $display("FAIL: Incorrect GPIO channel");
            errors = errors + 1;
        end

        if (errors == 0)
            $display("PASS: COMPLETE RTL INTEGRATION TEST");
        else
            $display("FAIL: Integration errors = %0d", errors);

        $display("----------------------------------");

        #100;
        $finish;
    end

    // Watchdog: prevent an endless simulation if a response is missing
    initial begin
        #1000000;

        if (rx_count < 3) begin
            $display("FAIL: Timeout waiting for UART response");
            $finish;
        end
    end

endmodule

module puf_uniqueness_tb;

    reg [239:0] periods_A;
    reg [239:0] periods_B;

    wire [44:0] response_A;
    wire [44:0] response_B;

    integer i;
    integer hamming_distance;
    integer uniqueness_x100;

    // Device A pairwise response
    pairwise_response device_A (
        .period_values(periods_A),
        .response(response_A)
    );

    // Device B pairwise response
    pairwise_response device_B (
        .period_values(periods_B),
        .response(response_B)
    );

    initial begin

        $display("==========================================");
        $display("TWO-DEVICE PUF UNIQUENESS TEST");
        $display("==========================================");

        // Simulated measurements for Device A
        periods_A[0*24 +: 24] = 24'd10;
        periods_A[1*24 +: 24] = 24'd15;
        periods_A[2*24 +: 24] = 24'd12;
        periods_A[3*24 +: 24] = 24'd23;
        periods_A[4*24 +: 24] = 24'd21;
        periods_A[5*24 +: 24] = 24'd30;
        periods_A[6*24 +: 24] = 24'd27;
        periods_A[7*24 +: 24] = 24'd43;
        periods_A[8*24 +: 24] = 24'd39;
        periods_A[9*24 +: 24] = 24'd47;

        // Simulated measurements for Device B
        periods_B[0*24 +: 24] = 24'd19;
        periods_B[1*24 +: 24] = 24'd55;
        periods_B[2*24 +: 24] = 24'd43;
        periods_B[3*24 +: 24] = 24'd13;
        periods_B[4*24 +: 24] = 24'd28;
        periods_B[5*24 +: 24] = 24'd25;
        periods_B[6*24 +: 24] = 24'd58;
        periods_B[7*24 +: 24] = 24'd16;
        periods_B[8*24 +: 24] = 24'd37;
        periods_B[9*24 +: 24] = 24'd49;

        #10;

        $display("\nDEVICE A PERIODS:");

        for (i = 0; i < 10; i = i + 1)
            $display("GPIO %0d = %0d cycles",
                i, periods_A[i*24 +: 24]);

        $display("\nDEVICE B PERIODS:");

        for (i = 0; i < 10; i = i + 1)
            $display("GPIO %0d = %0d cycles",
                i, periods_B[i*24 +: 24]);

        $display("\nDEVICE A RESPONSE:");
        $display("%045b", response_A);

        $display("\nDEVICE B RESPONSE:");
        $display("%045b", response_B);

        // Calculate Hamming distance
        hamming_distance = 0;

        for (i = 0; i < 45; i = i + 1) begin
            if (response_A[i] != response_B[i])
                hamming_distance = hamming_distance + 1;
        end

        // Calculate normalized difference in hundredths of a percent
        uniqueness_x100 =
            (hamming_distance * 10000 + 22) / 45;

        $display("\n==========================================");
        $display("UNIQUENESS ANALYSIS");
        $display("==========================================");

        $display("Total response bits = 45");
        $display("Different bits = %0d", hamming_distance);
        $display("Matching bits = %0d",
                 45 - hamming_distance);

        $display("Hamming distance = %0d",
                 hamming_distance);

        $display("Inter-device difference = %0d.%02d%%",
                 uniqueness_x100 / 100,
                 uniqueness_x100 % 100);

        $display("==========================================");

        if (hamming_distance >= 0 &&
            hamming_distance <= 45)
            $display("PASS: Hamming distance calculated");
        else
            $display("FAIL: Invalid Hamming distance");

        $display("TWO-DEVICE COMPARISON COMPLETED");
        $display("==========================================");

        $finish;

    end

endmodule

`timescale 1ns / 1ps

module uart_controller_tb;

    reg clk;
    reg rst;

    reg [7:0] rx_data;
    reg rx_dv;

    wire [3:0] channel_select;

    wire req_meas;
    reg [23:0] period_data;
    reg ack_meas;

    wire [7:0] tx_data;
    wire tx_start;
    reg tx_busy;
    reg tx_done;

    integer tx_count;
    integer errors;
    integer delay_count;

    uart_controller uut (
        .clk(clk),
        .rst(rst),

        .rx_data(rx_data),
        .rx_dv(rx_dv),

        .channel_select(channel_select),

        .req_meas(req_meas),
        .period_data(period_data),
        .ack_meas(ack_meas),

        .tx_data(tx_data),
        .tx_start(tx_start),
        .tx_busy(tx_busy),
        .tx_done(tx_done)
    );

    always #50 clk = ~clk;

    // Send a simulated received UART byte to the controller
    task send_command_byte;
        input [7:0] data;
        begin
            @(negedge clk);
            rx_data = data;
            rx_dv = 1'b1;

            @(negedge clk);
            rx_dv = 1'b0;
        end
    endtask

    // Simulated transmitter handshake
    // This verifies controller sequencing, not the serial waveform.
    always @(posedge clk) begin
        if (rst) begin
            tx_busy <= 1'b0;
            tx_done <= 1'b0;
            delay_count <= 0;
            tx_count <= 0;
        end
        else begin
            tx_done <= 1'b0;

            if (tx_start && !tx_busy) begin
                tx_busy <= 1'b1;
                delay_count <= 5;

                tx_count <= tx_count + 1;

                $display("Controller sends byte %0d = 0x%02h",
                         tx_count + 1, tx_data);

                if (tx_count == 0 && tx_data !== 8'h12)
                    errors <= errors + 1;

                if (tx_count == 1 && tx_data !== 8'h34)
                    errors <= errors + 1;

                if (tx_count == 2 && tx_data !== 8'h56)
                    errors <= errors + 1;
            end

            else if (tx_busy) begin
                if (delay_count == 0) begin
                    tx_busy <= 1'b0;
                    tx_done <= 1'b1;
                end
                else begin
                    delay_count <= delay_count - 1;
                end
            end
        end
    end

    initial begin
        clk = 0;
        rst = 1;

        rx_data = 0;
        rx_dv = 0;

        period_data = 24'h123456;
        ack_meas = 0;

        tx_busy = 0;
        tx_done = 0;

        tx_count = 0;
        errors = 0;
        delay_count = 0;

        $display("----------------------------------");
        $display("UART CONTROLLER TEST");
        $display("----------------------------------");

        repeat (5) @(negedge clk);
        rst = 0;

        // Send measurement command
        $display("Sending command: 10");
        send_command_byte(8'd10);

        // Send GPIO selection
        $display("Sending GPIO selection: 7");
        send_command_byte(8'd7);

        wait (req_meas == 1'b1);

        if (channel_select == 4'd7)
            $display("PASS: GPIO 7 selected");
        else begin
            $display("FAIL: Incorrect GPIO selection");
            errors = errors + 1;
        end

        // Simulate period measurement completion
        $display("Measurement started");

        repeat (20) @(negedge clk);
        ack_meas = 1'b1;

        @(negedge clk);
        ack_meas = 1'b0;

        wait (tx_count == 3);
        wait (tx_done == 1'b1);
        @(negedge clk);

        if (errors == 0)
            $display("PASS: UART controller test completed");
        else
            $display("FAIL: Controller test errors = %0d", errors);

        $display("----------------------------------");

        #100;
        $finish;
    end

endmodule

`timescale 1ns / 1ps

module uart_tx_tb;

    reg clk;
    reg rst;
    reg [7:0] tx_data;
    reg tx_start;

    wire tx_serial;
    wire tx_busy;
    wire tx_done;

    localparam integer BIT_TIME = 8700;

    integer byte_count;

    uart_tx uut (
        .clk(clk),
        .rst(rst),
        .tx_data(tx_data),
        .tx_start(tx_start),
        .tx_serial(tx_serial),
        .tx_busy(tx_busy),
        .tx_done(tx_done)
    );

    // 10 MHz clock
    always #50 clk = ~clk;

    // Send one byte and wait for completion
    task transmit_byte;
        input [7:0] data;
        begin
            @(negedge clk);
            tx_data = data;
            tx_start = 1'b1;

            @(negedge clk);
            tx_start = 1'b0;

            wait (tx_done == 1'b1);
            #1;

            byte_count = byte_count + 1;

            $display("Byte %0d transmitted = 0x%02h",
                     byte_count, data);

            @(negedge clk);
        end
    endtask

    initial begin
        clk = 0;
        rst = 1;
        tx_data = 0;
        tx_start = 0;
        byte_count = 0;

        $display("----------------------------------");
        $display("UART TRANSMITTER TEST");
        $display("----------------------------------");

        repeat (5) @(negedge clk);
        rst = 0;

        // Example 24-bit measurement: 0x123456
        $display("Transmitting 24-bit value: 0x123456");

        transmit_byte(8'h12);
        transmit_byte(8'h34);
        transmit_byte(8'h56);

        if (byte_count == 3)
            $display("PASS: All three bytes transmitted");

        $display("----------------------------------");
        #100;
        $finish;
    end

endmodule
